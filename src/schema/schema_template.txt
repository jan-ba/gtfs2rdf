/*
================================================================================
GTFS2RDF — Schema Template (NOT COMPILED)
================================================================================

Drop this file somewhere that is NOT picked up by your build (e.g. src/schema/schema_template.txt).
Copy-paste it into a new *.cppm schema file when you want to add a schema.

This template follows your current GTFS2RDF "schema file" structure:
  - one exported build<Name>Schema(...) function that returns schema::Schema
  - a list of possible_columns (for header validation / column lookup)
  - a prefixes map (used for Turtle prefixes and CURIE expansion)
  - a vector of rdf::Triple entries with placeholder instructions


-------------------------------------------------------------------------------
1) Placeholder / transform syntax inside strings
-------------------------------------------------------------------------------
In any instruction string, `{...}` marks a placeholder that is filled from the GTFS row.

A placeholder has this general form:

    { field1,field2,... | transformA | transformB | ... }

Rules:
  - whitespace inside the braces is removed, so `{ stop_lat , stop_lon | foo }`
    behaves like `{stop_lat,stop_lon|foo}`.
  - the part BEFORE the first '|' is a comma-separated list of column names
    (arguments). See field_transforms::TransformRegistry::parse_placeholder_with_functors().
  - transforms are chained with '|'. (MaxTransforms is currently 3.)
  - MaxArgs is currently 9 (covers e.g. weekday flags in calendar.txt).

-------------------------------------------------------------------------------
2) Transform kinds: Transform2One vs Transform2N
-------------------------------------------------------------------------------
Transform2One:
  - signature: void(const ArgSpan& args, std::string& out)
  - produces a single string output

Transform2N:
  - signature: void(const ArgSpan& args, std::vector<std::string>& out)
  - produces multiple outputs, and GTFS2RDF will duplicate the rendered instruction
    across multiple output "rows" (one per element).

Important limitation (current implementation):
  - within ONE instruction string (subject AND predicate AND object), only ONE
    Transform2N may appear in total.

Also note the evaluation order:
  - transforms BEFORE a Transform2N may receive ArgSpan(args...), a subsequent Transform2N
    has to have only one argument then 
  - transforms AFTER a Transform2N are applied to EACH produced element

-------------------------------------------------------------------------------
3) Triple structure
-------------------------------------------------------------------------------
A triple is: { SUBJECT, PREDICATE, OBJECT }

  - SUBJECT / PREDICATE are rdf::IRI
  - OBJECT is an rdf::Object, which can be:
      * a plain literal:           { "{field}" }
      * a language-tag literal:    { "{field}", "de" }
      * a typed literal:           { "{field}", IRI("xsd","integer") }
      * an IRI object:             { IRI("gtfs","Stop") }

-------------------------------------------------------------------------------
4) Where to register transforms
-------------------------------------------------------------------------------
You can:
  - register transforms inside build<Name>Schema(...) (recommended for schema-local transforms)
  - or put shared transforms into a common module (e.g. transform_lib) and register them here

Example below shows both a Transform2One (single output) and a Transform2N (multi output).
================================================================================
*/

namespace example_transforms {

// Transform2One example (single output).
// - Uses 2 args: lon, lat -> WKT POINT(lon lat)
// - Demonstrates MULTI-ARG transform usage in placeholders: {lon,lat|wkt_point}
void wkt_point_lon_lat(const field_transforms::ArgSpan& args, std::string& out) {
  if (args.size() != 2) {
    throw std::runtime_error("wkt_point_lon_lat: expected 2 args (lon,lat)");
  }
  const auto& lon = args[0];
  const auto& lat = args[1];
  if (lon.empty() || lat.empty()) {
    // Up to you: throw (hard fail) or return empty (skip triple upstream)
    throw std::runtime_error("wkt_point_lon_lat: empty lon/lat");
  }
  out = "POINT(" + lon + " " + lat + ")";
}

// Transform2N example (multi output).
// - Takes 1 arg and splits on ';' into multiple outputs.
// - Demonstrates MULTI-ROW expansion via placeholders: {field|split_semicolon}
void split_semicolon(const field_transforms::ArgSpan& args,
                            std::vector<std::string>& out) {
  if (args.size() != 1) {
    throw std::runtime_error("split_semicolon: expected 1 arg");
  }
  const std::string& s = args[0];
  std::string cur;
  for (char ch : s) {
    if (ch == ';') {
      if (!cur.empty()) out.push_back(cur);
      cur.clear();
    } else {
      cur.push_back(ch);
    }
  }
  if (!cur.empty()) out.push_back(cur);

  // Optionally: if you want “no values” to mean “no triples”, just leave out empty.
  // If you want to keep an empty marker, push_back("") instead.
}

} // namespace example_transforms


// =============================================================================
// 2) REQUIRED: build function
// =============================================================================
//
// Naming convention (required by build system):
//   - module file:  src/schema/your_schema_name.cppm
//   - build symbol: schema::buildYourSchemaNameSchema(registry)

export Schema buildYourSchemaNameSchema(field_transforms::TransformRegistry& registry) {

  // ---------------------------------------------------------------------------
  // (A) Register transforms (schema-local + shared)
  // ---------------------------------------------------------------------------
  // Register schema-local transforms:
  registry.registerTransform("wkt_point", example_transforms::wkt_point_lon_lat);
  registry.registerTransform("split_semicolon", example_transforms::split_semicolon);


  // ---------------------------------------------------------------------------
  // (B) Columns that may appear in the GTFS file header
  // ---------------------------------------------------------------------------
  // These are the columns your schema can reference in placeholders.
  // If a column referenced in a triple is missing in the input header, the triple
  // will be skipped for that file (see schema::Instruction validation).
  const std::vector<std::string> possible_columns = {
    // Required/commonly present:
    "id",
    "name",

    // Example for multi-arg transform:
    "lon",
    "lat",

    // Example for multi-row expansion:
    "tags"
  };

  // ---------------------------------------------------------------------------
  // (C) Prefixes (Turtle expansion)
  // ---------------------------------------------------------------------------
  const std::unordered_map<std::string, std::string> prefixes = {
    { "ex",   "https://example.org/resource/" },
    { "rdf",  "http://www.w3.org/1999/02/22-rdf-syntax-ns#" },
    { "xsd",  "http://www.w3.org/2001/XMLSchema#" },
    { "gtfs", "https://w3id.org/gtfs2rdf#" }
  };

  // ---------------------------------------------------------------------------
  // (D) Subject IRI template
  // ---------------------------------------------------------------------------
  // You can embed placeholders in the IRI local part.
  // Example: ex:{id}
  const IRI subject = IRI("ex", "{id}");

  // ---------------------------------------------------------------------------
  // (E) Triples
  // ---------------------------------------------------------------------------
  const std::vector<Triple> triples = {
    // --- Type triple
    { subject, {"rdf","type"}, { IRI("gtfs","YourClass") } },

    // --- Plain literal
    { subject, {"gtfs","name"}, { "{name}" } },

    // --- Typed literal (e.g., xsd:decimal)
    // { subject, {"gtfs","longitude"}, { "{lon}", IRI("xsd","decimal") } },

    // --- Multi-arg placeholder + Transform2One
    // Produces: "POINT(lon lat)" as a literal (use geo:wktLiteral if you have it)
    // TODO: mention that there is a more efficient way for this
    { subject, {"gtfs","wktPoint"}, { "{lon,lat|wkt_point}" } },

    // --- Multi-row expansion via Transform2N
    // If tags = "a;b;c", this expands into THREE triples:
    //   ex:{id} gtfs:tag "a" .
    //   ex:{id} gtfs:tag "b" .
    //   ex:{id} gtfs:tag "c" .
    //
    // Note: at most ONE Transform2N may appear in this whole *instruction string*.
    { subject, {"gtfs","tag"}, { "{tags|split_semicolon}" } },
  };

  // ---------------------------------------------------------------------------
  // (F) Build Schema object
  // ---------------------------------------------------------------------------
  // First argument is the GTFS file name this schema targets.
  // (Use the real filename, e.g. "stops.txt", "trips.txt", etc.)
  Schema sc("your_schema_name.txt", possible_columns, prefixes, triples, registry);
  return sc;
}

} // namespace schema
